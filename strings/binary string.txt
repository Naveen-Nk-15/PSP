class StringBinary{
    public static int binarySubstring(int a, String str)
    {
        // Your code here
        char s[] = str.toCharArray();
        int k=0;
        for(int i=0;i<a;i++)
        {
            if(s[i] == '1')
            {
                k++;
            }
        }
        return k*(k-1)/2;
    }
}

class GfG
{
    int maxLen(int arr[], int n)
    {
        // Your code here
        HashMap<Integer,Integer> hm = new HashMap<Integer,Integer>(); 
        int sum=0;
        int maxlen=0;
        for(int i=0;i<n;i++)
        {
            sum+=arr[i];
            if(arr[i]==0 && maxlen==0)
            {
                maxlen=1;
            }
            
            if(sum==0)
            {
                maxlen=i+1;
            }
            
            Integer k = hm.get(sum);
            if(k!=null)
            {
                maxlen = Math.max(maxlen,i-k);
            }
            else{
                hm.put(sum,i);
            }
        }
        return maxlen;
    }
}

    HashMap <Integer,Integer> hm1 = new HashMap<Integer,Integer>();

        }
        else
        {
            hm.put(A1[i],1);
        }
    }

        }
        else
        {
            hm1.put(A2[i],1);
        }
        if(hm.containsKey(A2[i]))
        {
            for(int j=0;j<hm.get(A2[i]);j++,k++)

                x[k] = A2[i];

        }
        
    }

    {
        if(hm1.containsKey(A1[i]))
        {

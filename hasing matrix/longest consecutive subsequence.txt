class Subseq
{   
    // Function to find Longest Consecutive Subsequence
	static int findLongestConseqSubseq(int a[], int n)
	{
	    // HashSet to store numbers
		HashSet<Integer> hs = new HashSet<Integer>();
		int ans = 0;
		
		// Adding elements to hash
		for(int i=0; i<n; i++)
			hs.add(a[i]);
		
		for(int i=0; i<n; i++)
		{
			if(!hs.contains(a[i]-1))
			{
				int j = a[i];
				while(hs.contains(j))
				{
					j++;
				}
				if(ans < j-a[i])
					ans = j-a[i];
			}
		}
		return ans;
	}
}

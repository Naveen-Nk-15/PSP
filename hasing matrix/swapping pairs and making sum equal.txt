class Solution
{
    long findSwapValues(long a[], int n, long  b[], int m)
    {
        // Your code goes here
        Arrays.sort(a);
        Arrays.sort(b);
        int sum1=0;
        int sum2=0;
        for(int i=0;i<n;i++)
        {
            sum1+=a[i];
        }
        for(int i=0;i<m;i++)
        {
            sum2+=b[i];
        }
        if ((sum1-sum2)%2 !=0)
        {
            return -1;
        }
        int dif = (sum1-sum2)/2;
        int i=0,j=0;
        while(i<n && j<m)
        {
            int k=(int) a[i]-(int)b[j];
            if(k==dif)
            {
                return 1;
            }
            else if (k<dif)
            {
                i++;
            }
            else
            {
            
                j++;
            }
        }
        return -1;
    }
}

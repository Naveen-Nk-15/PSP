class GFG
{
    static void rotate(int arr[][]) 
    {
        // Code Here
        // System.out.println(matrix[0].length);
         for (int i = 0; i < arr.length; i++)
            for (int j = i; j < arr[0].length;j++) {
                int temp = arr[j][i];
                arr[j][i] = arr[i][j];
                arr[i][j] = temp;
            }
            
        for (int i = 0; i < arr[0].length; i++)
            for (int j = 0, k = arr[0].length - 1;j < k; j++, k--) {
                int temp = arr[j][i];
                arr[j][i] = arr[k][i];
                arr[k][i] = temp;
            }
    }
}

#include <stdio.h>
#include <string.h>
#include <math.h>
#include <stdlib.h>
// finding number of islands
int a[30][30],
    n,k,
    count=0;
void island(int i,int j)
{
    if (a[i][j]==0)
    {
        return ;
    }
    else
    {
        a[i][j]=0;
        if (i-1>=0 && j-1 >=0)
        { 
            island(i-1,j-1);
        }
        if (i-1>=0 && j >=0)
        {
            island(i-1,j);
        }
        if (i-1>=0 && j+1 <=k)
        {
            island(i-1,j+1);
        }
        if (j-1 >=0)
        {
            island(i,j-1);
        }
        if (j+1 <=k)
        {
            island(i,j+1);
        }
        if (i+1<=n && j-1 >= 0)
        {
            island(i+1,j-1);
        }
        if (i+1<=n && j <=k)
        {
            island(i+1,j);
        }
        if (i+1<=n && j+1 <=k)
        {
            island(i+1,j+1);
        }
    }
}
int main() {											 
    
    scanf("%d %d",&n,&k);
    for (int i=0;i<n;i++)
    {
        for(int j=0;j<k;j++)
        {
            scanf("%d",&a[i][j]);   
        }
    }
    for (int i=0;i<n;i++)
    {
        for(int j=0;j<k;j++)
        {
            if (a[i][j]==1)
            {
                island( i, j);
                count++;
                printf("%d %d\n",i,j);
            }           
        }
    }
    printf("%d",count);
    return 0;
}

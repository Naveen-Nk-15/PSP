def NumberofElementsInIntersection(a, b, n, m):
    #return: expected length of the intersection array.
    
    #code here
    count=0
    
    if n> m: #used if to run the lesser length loop
        a=set(a)  #converted set to remove duplicates
        for i in b:
            if i in a:
                count+=1
                a.remove(i)
    else:
        b=set(b)
        for i in a:
            if i in b:
                count+=1
                b.remove(i)
    return count

class Solution
{ 
    public static int print2largest(int a[],int n) 
    {
        //CODE HERE
        int max=-1,smax=-1;
        for(int i=0;i<n;i++)
        {
            if(a[i]>max)
            {
                smax=max;
                max=a[i];
            }
            if(a[i]!=max && a[i]>smax)
            {
                smax=a[i];
            }
        }
        return smax;
    }
}

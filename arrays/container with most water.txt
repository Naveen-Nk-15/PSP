{    
	int l = 0;
        int r = len -1;
        int area = 0;
    
        while (l < r)
        {
            // minimum length isfound so that water only fills upto that level
		// r-l is to find breadth area of rectangle = l/b 
            area = Math.max(area, Math.min(A[l], A[r]) * (r - l));
                        
            if (A[l] < A[r])
                l += 1;
                
            else
                r -= 1;
        }
        return area;   
    }

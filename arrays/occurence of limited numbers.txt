class Frequency{
    public static void frequencycount(int arr[], int n)
    {
        // code here
        for(int i=0;i<n;i++)
        {
            arr[i]=arr[i]-1;
        }
        for(int i=0;i<n;i++)
        {
            arr[arr[i]%n]=arr[arr[i]%n]+n;
        }
        for(int i=0;i<n;i++)
        {
            // System.out.println(arr[i]/n+"mysol");
            arr[i]=arr[i]/n;
        }
        
    }
}

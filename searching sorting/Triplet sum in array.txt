class TripletSum
{
    // arr[] : The input Array
    // N : Size of the Array
    // X : Sum which you need to search for 
    
    public static int find3Numbers(int arr[], int n, int X) { 
    
       // Your code Here
        Arrays.sort(arr);
        for(int i=0;i<n-2;i++)
        {
            int l=i+1;
            int r=n-1;
            while(l<r)
            {
                int sum = arr[i]+arr[l]+arr[r];
                // System.out.println(arr[i]+" "+l+" "+r+" "+sum);
                // if(r==i)
                if(sum==X)
                {
                    // System.out.println(arr[i]+" "+l+" "+r+" "+sum);
                    return 1;
                }
                if(sum>X)
                {
                    r--;
                }
                else
                {
                    l++;
                }
            }
        }
        return 0;
    }
}

class Solution
{
    static int findNumberOfTriangles(int a[], int n)
    {
        // code here
        int count=0;
        Arrays.sort(a);
        for(int i=n-1;i>=1;i--)
        {
            int l=0 ;
            int r=i-1;
            while(l<=r)
            {
                if(a[l]+a[r]>a[i])
                {
                    count+=r-l;
                    r--;
                }
              else
                {
                    l++;
                }
                
            }
        }
        return count;
    }
}

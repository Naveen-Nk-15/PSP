Merge sort

class GfG
{
    void merge(int arr[], int l, int m, int r)
    {
         // Your code here
        int n2 = r-m;
        int n1 = m-l+1;
        int t1 [] = new int[n1];
        int t2 [] = new int[n2];
        
        for(int i=0;i<n1;i++)
        {
            t1[i] = arr[l+i];
        }
        for(int i=0;i<n2;i++)
        {
            t2[i] = arr[m+1+i];
        }
        int i=0;
        int j=0;
        int k=l;
        while(i<n1&&j<n2)
        {
            if(t1[i]<=t2[j])
            {
                arr[k]=t1[i];
                i++;
            }
            else
            {
                arr[k]=t2[j];
                j++;
            }
            k++;
        }
        while(i<n1)
        {
            arr[k]=t1[i];
            k++;i++;
        }
        while(j<n2)
        {
            arr[k]=t2[j];
            k++;j++;
        }
    }
}

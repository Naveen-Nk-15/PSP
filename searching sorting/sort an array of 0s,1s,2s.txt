sort an array 0s,1s and 2s


public static void sort012(int a[], int n){
    // code here 
    int low=0,mid=0,high=n-1;
    while(mid<=high)
    {
        if (a[mid]==0)
        {
            int t=a[low];
            a[low]=a[mid];
            a[mid]=t;
            low++;mid++;
        }
        else if(a[mid]==2)
        {
            int t=a[mid];
            a[mid]=a[high];
            a[high]=t;
            high--;
        }
        else{
            mid++;
        }
    }
}
